<script>
const API = '/api/status';
const $ = s => document.querySelector(s);
const esc = s => String(s ?? '').replace(/[&<>"']/g, c => ({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'}[c]));

function monthName(ym) {
  const [y, m] = ym.split('-').map(Number);
  return new Date(y, m - 1, 1).toLocaleDateString('en-US', { month: 'long', year: 'numeric' });
}
function daysText(n) {
  if (n <= 7) return n === 1 ? '1 day' : `${n} days`;
  const w = Math.round(n / 7);
  return w === 1 ? '1 week' : `${w} weeks`;
}
function rangeText(lo, hi) {
  if (lo === 0) return `within ${daysText(hi)}`;
  const a = daysText(lo), b = daysText(hi);
  if (a === b) return `in about ${a}`;
  // "2 to 5 weeks" instead of "2 weeks to 5 weeks"
  const [an, au] = a.split(' '), [bn, bu] = b.split(' ');
  return au.replace(/s$/, '') === bu.replace(/s$/, '') ? `in ${an} to ${bn} ${bu}` : `in ${a} to ${b}`;
}

function renderAward(a, contact) {
  const k = a.stage.kind;
  let html = `<section class="card"><h2>${esc(a.awardName)}</h2>`;

  if (k === 'done') {
    html += `<p class="stage-line">Status: <strong>Complete</strong></p><p>${esc(a.stage.description)}</p>`;
  } else if (k === 'closed') {
    html += `<p class="stage-line">Status: <strong>Closed</strong></p><p>${esc(a.stage.description)}</p>`;
  } else if (k === 'waitlist') {
    html += `<p class="stage-line">Status: <strong>Waitlist</strong></p>
      <div class="box"><p class="big">You're on the waitlist</p>
      <p class="small">${esc(a.stage.description)} We can't give a time estimate while you're on the waitlist.</p></div>`;
  } else {
    html += `<p class="stage-line">Current step: <strong>${esc(a.stage.publicName)}</strong>`
      + (a.inStageSinceDays !== null ? ` &middot; ${a.inStageSinceDays === 0 ? 'started today' : `started ${daysText(a.inStageSinceDays)} ago`}` : '')
      + `</p><div class="grid">`;

    if (a.waitingFor) {
      const you = a.waitingFor.key === 'Applicant';
      html += `<div class="box ${you ? 'you' : ''}"><h3>Waiting on</h3>
        <p class="big">${esc(a.waitingFor.label)}${you ? '<span class="pill">Action needed</span>' : ''}</p>
        <p class="small">${esc(a.waitingFor.message)}</p></div>`;
    }
    if (a.nextStep) {
      html += `<div class="box"><h3>Next step: ${esc(a.nextStep.name)}</h3>`;
      html += a.nextStep.overdue
        ? `<p class="big">Taking longer than usual</p><p class="small">${esc(a.stage.publicName)} usually takes about ${daysText(a.nextStep.typicalStageDays)} in total.</p>`
        : `<p class="big">Usually ${rangeText(a.nextStep.lowDays, a.nextStep.highDays)}</p><p class="small">${esc(a.stage.publicName)} usually takes about ${daysText(a.nextStep.typicalStageDays)} in total.</p>`;
      html += `</div>`;
    } else if (k === 'closeout') {
      html += `<div class="box"><h3>Next step</h3><p class="big">Closeout</p><p class="small">Once we have the remaining items, your award is complete.</p></div>`;
    }
    html += `</div>`;

    if (a.completion) {
      html += `<p>Estimated completion: <strong>around ${esc(monthName(a.completion.approxDate))}</strong></p>`;
    }
    if (a.note) html += `<p class="msg note">${esc(a.note)}</p>`;
  }

  if (k !== 'closed' && k !== 'waitlist') {
    html += `<h3 class="sr">All steps</h3><ol class="steps" style="margin-top:20px">`;
    a.steps.forEach((name, i) => {
      const cls = i < a.currentStepIndex ? 'done' : i === a.currentStepIndex ? 'current' : 'todo';
      const mark = cls === 'done' ? '&#10003;' : i + 1;
      const label = cls === 'done' ? 'done' : cls === 'current' ? 'current step' : 'upcoming';
      html += `<li class="${cls}"><span class="dot" aria-hidden="true">${mark}</span><span class="step-name">${esc(name)} <span class="sr">(${label})</span></span></li>`;
    });
    html += `</ol>`;
  }
  return html + `</section>`;
}

$('#lookup').addEventListener('submit', async e => {
  e.preventDefault();
  const companyName = $('#company').value.trim();
  const email = $('#email').value.trim();
  const out = $('#results');
  if (!companyName || !email) {
    out.innerHTML = `<div class="card"><p class="msg">Please enter your company name and email address.</p></div>`;
    return;
  }
  const btn = $('#go');
  btn.disabled = true; btn.textContent = 'Checking...';
  try {
    const res = await fetch(API, { method: 'POST', headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ companyName, email }) });
    const data = await res.json();
    if (!res.ok) {
      out.innerHTML = `<div class="card"><p class="msg">${esc(data.error || 'Something went wrong. Please try again.')}</p></div>`;
    } else if (!data.found) {
      out.innerHTML = `<div class="card"><p class="msg">We couldn't find an award that matches. Check the spelling of your company name and use the email address you applied with. Still stuck? Email <a href="mailto:${esc(data.contact?.email)}">${esc(data.contact?.email)}</a> and we'll help.</p></div>`;
    } else {
      out.innerHTML = data.awards.map(a => renderAward(a, data.contact)).join('')
        + `<p class="small" style="margin-bottom:20px">Questions? Contact ${esc(data.contact.name)} at <a href="mailto:${esc(data.contact.email)}">${esc(data.contact.email)}</a>.</p>`;
    }
  } catch {
    out.innerHTML = `<div class="card"><p class="msg">We couldn't reach our records right now. Please try again in a few minutes.</p></div>`;
  } finally {
    btn.disabled = false; btn.textContent = 'Check status';
  }
});
</script>
